cmake_minimum_required(VERSION 3.20)
project(crossroad LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "ASan and UBSan" OFF)
add_compile_definitions(_POSIX_C_SOURCE=200809L)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(crossroad STATIC src/config.c src/model.c src/io.c src/net.c)
target_include_directories(crossroad PUBLIC src)
foreach(role server client observer)
  add_executable(crossroad_${role} src/${role}.c)
  target_link_libraries(crossroad_${role} PRIVATE crossroad)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(model_test tests/test_model.c)
  target_link_libraries(model_test PRIVATE crossroad)
  add_test(NAME model COMMAND model_test ${CMAKE_CURRENT_SOURCE_DIR}/data)
  set_tests_properties(model PROPERTIES TIMEOUT 10)
  add_executable(net_test tests/test_net.c)
  target_link_libraries(net_test PRIVATE crossroad)
  add_test(NAME protocol_buffers COMMAND net_test)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  add_test(NAME protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run.py
      --build-dir ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(protocol_buffers PROPERTIES TIMEOUT 10)
  set_tests_properties(protocol PROPERTIES TIMEOUT 40)
  add_test(NAME scenarios COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run.py
      --build-dir ${CMAKE_CURRENT_BINARY_DIR} --suite scenarios)
  set_tests_properties(scenarios PROPERTIES TIMEOUT 50)
  add_test(NAME observers COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run.py
      --build-dir ${CMAKE_CURRENT_BINARY_DIR} --suite observers)
  set_tests_properties(observers PROPERTIES TIMEOUT 40)
endif()
